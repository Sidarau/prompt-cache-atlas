API Usage

# Check cache before calling a model
curl -X POST "https://your-app.com/api/cache/check" \\
-H "Content-Type: application/json" \\
-d '{"prompt":"What are the 15 rules to reduce AI token consumption?","model":"claude-sonnet-4","threshold":0.95}'

# Response: {"cached":true,"tokens_saved":1550,"hit_count":9}

# Store a response for future cache hits
curl -X POST "https://your-app.com/api/cache/store" \\
-H "Content-Type: application/json" \\
-d '{"prompt":"...","response":"...","tokens_in":1200,"tokens_out":350}'

# Get stats
curl "https://your-app.com/api/stats"
# Returns: hit_rate, tokens_saved, cost_saved, top_cached
